+                        chain_indices_in_range_from_implies_index(
+                            witness@,
+                            attestations@,

+                        );
+                    }
+                    digest_equal(&attestations[witness[witness.len() - 1]].subject.digest, &subject.digest)
+                }
+        }
+    }
